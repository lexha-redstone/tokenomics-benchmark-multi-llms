# Third-party packages ClassEval's own tasks and tests import.
#
# These are NOT optional extras. A package that is missing does not make a task
# unscorable -- it makes the machine unscorable, and `tools/classeval_preflight.py`
# would otherwise quarantine the task and quietly shrink the benchmark. Two
# machines with different packages installed measure different task sets, so
# their pass rates cannot be compared. Install these before the first sweep.
#
# Derived from the dataset, not hand-listed: see `classeval_required_modules()`
# in src/datasets.py, and re-derive after any split refresh with
#   python3 tools/classeval_preflight.py --deps-only
#
# package            import name    tasks
numpy              # numpy         ClassEval_30, _51, _57, _59, _93
pandas             # pandas        ClassEval_28
gensim             # gensim        ClassEval_44, _93
beautifulsoup4     # bs4           ClassEval_44
Pillow             # PIL           ClassEval_45
python-docx        # docx          ClassEval_34
openpyxl           # openpyxl      ClassEval_38
PyPDF2             # PyPDF2        ClassEval_69
reportlab          # reportlab     ClassEval_69
nltk               # nltk          ClassEval_52

# nltk also needs data files, which pip does not install. Do NOT hardcode the
# resource names: they moved between nltk releases (`pos_tag` wanted
# `averaged_perceptron_tagger` before 3.9 and `averaged_perceptron_tagger_eng`
# after; `word_tokenize` moved from `punkt` to `punkt_tab`), so any fixed list
# is wrong on some installed version. Ask the installed nltk instead:
#
#   python3 tools/fetch_nltk_data.py            # what is missing here
#   python3 tools/fetch_nltk_data.py --install  # fetch it
#
# That fetches the package zips over ordinary HTTPS and so honours HTTPS_PROXY.
# `nltk.download()` refuses to run behind a proxy at all -- it cannot pin the
# validated IP when a proxy performs the egress, and fails with
# "Security Violation [pathsec.urlopen] ... (CWE-918)".
